## Glossary

- **Strike agent** — a work agent spawned by `pan strike <id>` running in `workspaces/feature-<id>-strike` on branch `strike/<id>`. Its contract is defined in `roles/strike.md`.
- **Strike orchestrator** — whoever spawned the strike (Flywheel or operator). Per `roles/strike.md:73`, the orchestrator owns: reviewing the `strike/<id>` branch, merging it (via a squash-merge through GH API), and running `pan done <id> --strike`.
- **Kickoff prompt / `.pan/kickoff.md`** — the initial instruction delivered to a strike agent by `buildStrikePrompt()` in `src/cli/commands/strike.ts`, written to the workspace's `.pan/kickoff.md` by the delivery system.
- **`pan done --strike`** — the CLI command that runs `verifyStrikeBranchMergedIntoMain()` and then `postMergeLifecycle()`, marking the issue `verifying_on_main` without entering the review pipeline.
- **Squash-merge** — a GitHub merge mode that creates one new commit on `main` incorporating all branch changes, but does NOT fast-forward or replay the original branch commits. After a squash-merge, the original `strike/<id>` branch commits are NOT ancestors of `origin/main`.
- **`git merge-base --is-ancestor A B`** — exits with code 0 if A is an ancestor of B, code 1 otherwise. For a squash-merged branch, `strike/<id>` is NOT an ancestor of `origin/main` (the squash commit is new), so this always fails for the documented flow.
- **`getPullRequestState`** — the existing Effect function in `src/lib/github-app.ts:783` that returns `{ merged: boolean; state?: string }` for a PR by owner/repo/number; wraps a GH API call; used by merge-fulfillment already.
- **`pan close --force`** — the close command that already handles squash-merged branches correctly by checking PR merged-state rather than branch ancestry (verified by examining `src/cli/commands/close.ts`; see prior-art note in W2).

export async function verifyStrikeBranchMergedIntoMain(issueId: string, projectPath: string): Promise<string> {
  const branchName = `strike/${issueId.toLowerCase()}`;
  const issueLower = issueId.toLowerCase();

  // Fetch so local state reflects origin.
  await execAsync('git fetch origin main', {
    cwd: projectPath,
    encoding: 'utf-8',
    timeout: 60000,
  });

  // PAN-2300: ancestry check fails for squash-merges (the squash commit is new;
  // the original branch commits are NOT in origin/main's ancestry).
  // Primary verification: PR merged-state via GH API (works for all merge modes).
  // This matches how pan close --force detects squash-merged PRs (merge-fulfillment.ts).
  const { Effect } = await import('effect');
  const { isGitHubAppConfigured, getPullRequestState } = await import('../../lib/github-app.js');
  const { loadProjectsConfigSync, resolveProjectFromIssueSync } = await import('../../lib/projects.js');
  const { extractNumberSync } = await import('../../lib/issue-id.js');

  const resolved = resolveProjectFromIssueSync(issueId);
  if (resolved && isGitHubAppConfigured()) {
    const prNumber = await findStrikePrNumber(issueLower, projectPath);
    if (prNumber !== null) {
      const [owner, repo] = resolved.projectPath
        .replace(/.*\/([^/]+)$/, '$1')  // simple fallback; use config if available
        .split('/');
      // Use the full repo slug from project config if available.
      const repoConfig = loadProjectsConfigSync()?.projects?.find(
        p => p.path === resolved.projectPath
      );
      const repoOwner = repoConfig?.github?.owner ?? 'eltmon';
      const repoName = repoConfig?.github?.repo ?? resolved.projectPath.split('/').pop() ?? repo;
      const prState = await Effect.runPromise(getPullRequestState(repoOwner, repoName, prNumber));
      if (prState.merged) {
        return `${branchName} PR #${prNumber} was squash-merged into origin/main`;
      }
      throw new Error(
        `PR #${prNumber} for ${branchName} exists but was not merged (state: ${prState.state ?? 'unknown'}). ` +
        'Merge the PR before running pan done --strike.',
      );
    }
  }

  // Fallback: direct ancestry check (FF-merge case, or GH App not configured).
  const localBranchExists = await execAsync(`git rev-parse --verify ${shellQuote(branchName)}`, {
    cwd: projectPath,
    encoding: 'utf-8',
    timeout: 10000,
  }).then(() => true).catch(() => false);

  if (!localBranchExists) {
    throw new Error(
      `${branchName} not found locally and no merged PR was detected. ` +
      'If the branch was squash-merged and deleted, the GH App must be configured to verify via PR state.',
    );
  }

  await execAsync(`git merge-base --is-ancestor ${shellQuote(branchName)} origin/main`, {
    cwd: projectPath,
    encoding: 'utf-8',
    timeout: 10000,
  });

  return `${branchName} is contained in origin/main (fast-forward merge)`;
}

/** Find the PR number for a strike branch by querying the GH CLI. */
async function findStrikePrNumber(issueLower: string, projectPath: string): Promise<number | null> {
  try {
    const { stdout } = await execAsync(
      `gh pr list --head strike/${issueLower} --state all --json number --jq '.[0].number'`,
      { cwd: projectPath, encoding: 'utf-8', timeout: 15000 },
    );
    const n = parseInt(stdout.trim(), 10);
    return Number.isFinite(n) ? n : null;
  } catch {
    return null;
  }
}
